什麼是線段樹？

線段樹 (Segment Tree) 是一種儲存「區間」資訊的二元樹資料結構。當遇到需要頻繁進行區間查詢（如區間和、區間最大/最小值）與單點/區間修改時，線段樹能將複雜度降低至 \mathcal{O}(\log N)。

  • 根節點： 代表整個區間 $[0, N-1]$。
  • 葉子節點： 代表單一元素 $[i, i]$。
  • 內部節點： 代表左子樹與右子樹合起來的區間。
空間複雜度： 若陣列大小為 $N$，建議配置 $4N$ 的空間以確保不會溢位。

常見操作時間複雜度比較：

操作項目 暴力法 / 前綴和 線段樹 (Segment Tree)
單點修改 $\mathcal{O}(1)$ / $\mathcal{O}(N)$ $\mathcal{O}(\log N)$
區間查詢 $\mathcal{O}(N)$ / $\mathcal{O}(1)$ $\mathcal{O}(\log N)$
區間修改 (含 Lazy) $\mathcal{O}(N)$ $\mathcal{O}(\log N)$

進階技術：懶惰標記 (Lazy Propagation)

當我們需要進行區間修改（Range Update）時（例如：將區間 $[L, R]$ 內的所有數字都加上 $v$），若對每一個葉子節點逐一修改，複雜度會退化至 $\mathcal{O}(N \log N)$。

核心精神：「先記著，用到再更新」

當查詢區間完全涵蓋當前節點代表的區間時，直接修改該節點的值，並在該節點打上一個 lazy 標記，然後立刻返回！不繼續向下遞迴。

下放標記 (Push Down)

當之後的查詢或修改操作「不得不存取」該節點的子節點時，才順便把 lazy 標記傳給左右子節點，並清除當前節點的標記。

C++ 實作範例程式碼

這是最經典的點修改、區間和查詢線段樹範例，採用 1-based 索引進行陣列實作（左子節點為 2*p，右子節點為 2*p+1）。 

#include <iostream>
#include <vector>

using namespace std;

const int MAXN = 100005;
vector<long long> tree(4 * MAXN);
vector<long long> arr(MAXN);

// 建樹 (Build)
void build(int node, int l, int r) {
    if (l == r) {
        tree[node] = arr[l];
        return;
    }
    int mid = (l + r) / 2;
    build(2 * node, l, mid);          // 左子樹
    build(2 * node + 1, mid + 1, r);  // 右子樹
    tree[node] = tree[2 * node] + tree[2 * node + 1]; // PullUp
}

// 單點修改 (Point Update): 將 index qx 的值改為 val
void update(int node, int l, int r, int qx, long long val) {
    if (l == r) {
        tree[node] = val;
        return;
    }
    int mid = (l + r) / 2;
    if (qx <= mid)
        update(2 * node, l, mid, qx, val);
    else
        update(2 * node + 1, mid + 1, r, qx, val);
    tree[node] = tree[2 * node] + tree[2 * node + 1]; // 重新更新父節點
}

// 區間查詢 (Range Query): 查詢 [ql, qr] 的區間和
long long query(int node, int l, int r, int ql, int qr) {
    if (ql <= l && r <= qr) { // 完全涵蓋
        return tree[node];
    }
    int mid = (l + r) / 2;
    long long sum = 0;
    if (ql <= mid)
        sum += query(2 * node, l, mid, ql, qr);
    if (qr > mid)
        sum += query(2 * node + 1, mid + 1, r, ql, qr);
    return sum;
}

int main() {
    int n = 6;
    vector<int> input = {2, 5, 1, 4, 9, 3};
    for (int i = 0; i < n; i++) arr[i + 1] = input[i];

    build(1, 1, n);

    cout << "區間 [2, 5] 的和: " << query(1, 1, n, 2, 5) << endl; // 5+1+4+9 = 19
    update(1, 1, n, 3, 10); // 將索引 3 的值改為 10
    cout << "修改後區間 [2, 5] 的和: " << query(1, 1, n, 2, 5) << endl; // 5+10+4+9 = 28

    return 0;
}
